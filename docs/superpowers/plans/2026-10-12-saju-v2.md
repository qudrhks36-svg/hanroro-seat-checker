- 문장에서 "을(를)" 같은 조사 병기를 쓰지 않는다. 받침에 따라 달라지는 조사가 붙을 자리는 조사가 필요 없는 문장 구조로 쓴다.
- 커밋은 `git add` 로 `saju_web/src`, `saju_web/tests`와 필요한 삭제(`git rm`) 경로만 추가한다 (`git add -A` 금지: 저장소에 무관한 untracked 폴더가 많다). 커밋 메시지 끝에 `Co-Authored-By: Claude Sonnet 5.5 <noreply@anthropic.com>`.

## 계획에서 스펙과 다르게 정한 것 (Rulings)

- `reasons`는 "2개"가 아니라 **1~2개**다 (한쪽 요소가 부족하면 2개가 안 나온다).
- `buildCharacter`는 스펙의 `(pillars, analysis)`가 아니라 **명시적 입력 객체** `{dayGan, dayZhi, group, strength, dayElement}`를 받는다 (조합 전수 테스트가 쉬워진다).
- 스펙 결과 화면의 3번 "사주표와 오행 막대"는 별도 카드로 두지 않고, 1번 공유 카드 안의 사주표와 4번 "오행 균형" 풀이 카드의 막대(`meter`)로 충족한다 (같은 내용을 두 번 보여주지 않는다).
- 카드 문구(약 220개)는 계획에 일일이 싣지 않고 **규칙 + 예시 2개 + 완결성 테스트**로 정의해 구현자가 쓴다. 그 외 문구(MBTI 16문장, 말풍선 15문장, 캐릭터 표)는 계획에 전부 있다.
- MBTI `GAIN`(변환 배율) 조정은 분포 검증으로 정한다 (Task 1).

```ts
import data from '../data/mbti.json'
import type { Analysis } from './analysis'
import { ELEMENTS, ELEMENT_KO, GROUPS, isYang, type Element, type TenGodGroup } from './tables'

// ponytail: 가중치(0.5, 보너스 0.15/0.10)와 배율은 우리가 정한 해석 규칙이다. 검증된 사주-MBTI 대응표는 없다.
// 배율은 분포 검증(tests/mbti.test.ts)으로 정한다. 바꾸면 스펙의 샘플 표 값도 갱신한다.
export const MBTI_GAIN = 40
const MIN_PCT = 15
const MAX_PCT = 85

export type AxisId = 'EI' | 'SN' | 'TF' | 'JP'
export type AxisResult = {
  axis: AxisId
  letters: [string, string]
  pctFirst: number
  lean: string
  balanced: boolean
  reasons: { label: string; points: number }[]
}
export type MbtiPrediction = { axes: AxisResult[]; type: string }

type Side = { el: Element[]; gr: TenGodGroup[] }
const AXES: { id: AxisId; letters: [string, string]; a: Side; b: Side }[] = [
  { id: 'EI', letters: ['E', 'I'], a: { el: ['火', '木'], gr: ['비겁', '식상'] }, b: { el: ['水', '金'], gr: ['인성', '관성'] } },
  { id: 'SN', letters: ['S', 'N'], a: { el: ['土', '金'], gr: ['재성'] }, b: { el: ['木', '水'], gr: ['식상'] } },
  { id: 'TF', letters: ['T', 'F'], a: { el: ['金', '水'], gr: ['관성'] }, b: { el: ['火', '土'], gr: ['인성'] } },
  { id: 'JP', letters: ['J', 'P'], a: { el: ['土', '金'], gr: ['관성', '재성'] }, b: { el: ['木', '火'], gr: ['식상', '비겁'] } },
]

type Part = { label: string; v: number } // v > 0 이면 첫 글자(E/S/T/J) 쪽으로 미는 힘

export function predictMbti(a: Analysis): MbtiPrediction {
  const elTotal = ELEMENTS.reduce((s, e) => s + a.elements[e], 0) || 1
  const grTotal = GROUPS.reduce((s, g) => s + a.groups[g], 0) || 1
  const el = (e: Element) => a.elements[e] / elTotal
  const gr = (g: TenGodGroup) => a.groups[g] / grTotal
  const elLabel = (e: Element) => `${ELEMENT_KO[e]}(${e}) 기운`
  const yang = isYang(a.dayGan) ? 1 : -1

  const axes = AXES.map(({ id, letters, a: A, b: B }): AxisResult => {
    const parts: Part[] = [
      ...A.el.map((e) => ({ label: elLabel(e), v: 0.5 * el(e) })),
      ...A.gr.map((g) => ({ label: g, v: 0.5 * gr(g) })),
      ...B.el.map((e) => ({ label: elLabel(e), v: -0.5 * el(e) })),
      ...B.gr.map((g) => ({ label: g, v: -0.5 * gr(g) })),
    ]
    const dayLabel = `일간 오행(${ELEMENT_KO[a.dayElement]})`
    if (A.el.includes(a.dayElement)) parts.push({ label: dayLabel, v: 0.15 })
    if (B.el.includes(a.dayElement)) parts.push({ label: dayLabel, v: -0.15 })
    if (id === 'EI') parts.push({ label: '일간 음양', v: 0.1 * yang })

    const d = parts.reduce((s, p) => s + p.v, 0)
    const pctFirst = Math.max(MIN_PCT, Math.min(MAX_PCT, Math.round(50 + MBTI_GAIN * d)))
    const sign = pctFirst >= 50 ? 1 : -1
    const reasons = parts
      .filter((p) => p.v * sign > 0)
      .sort((x, y) => Math.abs(y.v) - Math.abs(x.v))
      .slice(0, 2)
      .map((p) => ({ label: p.label, points: Math.round(Math.abs(p.v) * MBTI_GAIN) }))
    return {
      axis: id,
      letters,
      pctFirst,
      lean: sign > 0 ? letters[0] : letters[1],
      balanced: Math.abs(pctFirst - 50) < 10,
      reasons,
    }
  })
  return { axes, type: axes.map((x) => x.lean).join('') }
}

export function axisText(axis: AxisId, letter: string): { advice: string; minor: string } {
  return (data as Record<string, Record<string, { advice: string; minor: string }>>)[axis][letter]
}
```

```ts
import data from '../data/character.json'
import type { Strength } from './analysis'
import type { Element, TenGodGroup } from './tables'

export type Character = {
  animal: string
  emoji: string
  job: string
  propEmoji: string
  propName: string
  adjective: string
  tone: string
  nickname: string
  accent: Element
  bubbles: { label: string; text: string }[]
}

const D = data as unknown as {
  gan: Record<string, { animal: string; emoji: string; job: string }>
  zhi: Record<string, { emoji: string; name: string }>
  adjective: Record<TenGodGroup, string>
  tone: Record<Strength, string>
  bubbles: Record<TenGodGroup, { love: string; work: string; stress: string }>
}

export function buildCharacter(i: {
  dayGan: string
  dayZhi: string
  group: TenGodGroup
  strength: Strength
  dayElement: Element
}): Character {
  const g = D.gan[i.dayGan]
  const z = D.zhi[i.dayZhi]
  const adjective = D.adjective[i.group]
  const b = D.bubbles[i.group]
  return {
    animal: g.animal,
    emoji: g.emoji,
    job: g.job,
    propEmoji: z.emoji,
    propName: z.name,
    adjective,
    tone: D.tone[i.strength],
    nickname: `${adjective} ${g.job} ${g.animal}`,
    accent: i.dayElement,
    bubbles: [
      { label: '연애할 때', text: b.love },
      { label: '일할 때', text: b.work },
      { label: '스트레스 받을 때', text: b.stress },
    ],
  }
}
```

```ts
function mergeLacking(cards: Card[]): Pick<Card, 'chips' | 'bullets'> {
  const per = Math.ceil(3 / cards.length)
  return {
    chips: [...new Set(cards.flatMap((c) => c.chips.slice(0, per)))].slice(0, 3),
    bullets: cards.flatMap((c) => c.bullets.slice(0, per)).slice(0, 3),
  }
}

export function interpret(p: Pillars, a: Analysis): Section[] {
  const s = D.strength[a.strength]
  const g = D.group[a.dominantGroup]
  const C = D.cards
  const gan = C.dayGan[a.dayGan]
  const str = C.strength[a.strength]
  const work = C.group[a.dominantGroup].work
  const yong = C.yongshin[a.yongshin]
  const elTotal = ELEMENTS.reduce((t, e) => t + a.elements[e], 0) || 1

  const lackCards = a.lacking.map((e) => C.lacking[e])
  const balance = lackCards.length ? mergeLacking(lackCards) : C.balanced
  const balanceSummary = !lackCards.length
    ? C.balanced.summary
    : lackCards.length === 1
      ? lackCards[0].summary
      : `${a.lacking.map(elementLabel).join('·')} 기운이 비어 있어요`

  return [
    {
      title: '기본 성향',
      summary: gan.summary,
      chips: gan.chips,
      bullets: [...gan.bullets.slice(0, 2), C.dayZhi[p.day.zhi].summary],
      more: [D.dayGan[a.dayGan], D.dayZhi[p.day.zhi]],
    },
    {
      title: '사주 구조',
      summary: str.summary,
      chips: str.chips,
      bullets: [...str.bullets.slice(0, 2), C.group[a.dominantGroup].trait.summary],
      more: [s.summary, g.trait],
      meter: [{ label: '자기 기운', value: Math.round(a.ratio * 100) }],
    },
    {
      title: '오행 균형',
      summary: balanceSummary,
      chips: balance.chips,
      bullets: balance.bullets,
      more: a.lacking.length ? a.lacking.map((e) => D.lacking[e]) : [D.balanced],
      meter: ELEMENTS.map((e) => ({ label: `${ELEMENT_KO[e]}(${e})`, value: Math.round((a.elements[e] / elTotal) * 100) })),
    },
    {
      title: '직업·활동 경향',
      summary: work.summary,
      chips: work.chips,
      bullets: work.bullets,
      more: [g.work],
    },
    {
      title: '조언',
      summary: yong.summary,
      chips: yong.chips,
      bullets: yong.bullets,
      more: [s.advice, D.yongshin[a.yongshin]],
    },
  ]
}
```

```ts
import type { MbtiPrediction } from './mbti'

export type Compare = { matched: string[]; onlySaju: string[]; onlyMbti: string[]; score: number; notes: string[] }

export function compareMbti(pred: MbtiPrediction, actual: string): Compare {
  if (!/^[EI][SN][TF][JP]$/.test(actual)) throw new Error(`invalid MBTI: ${actual}`)
  const matched: string[] = []
  const onlySaju: string[] = []
  const onlyMbti: string[] = []
  const notes: string[] = []
  pred.axes.forEach((x, i) => {
    const letter = actual[i]
    if (letter === x.lean) {
      matched.push(letter)
      return
    }
    onlySaju.push(x.lean)
    onlyMbti.push(letter)
    const pct = letter === x.letters[0] ? x.pctFirst : 100 - x.pctFirst
    notes.push(pct >= 35 ? `사주에서도 ${letter} 쪽이 ${pct}%로 꽤 있는 편입니다` : `${letter} 쪽은 사주 경향과 다른 축입니다`)
  })
  return { matched, onlySaju, onlyMbti, score: matched.length / 4, notes }
}
```

`saju_web/src/components/ShareCard.tsx` 전체 교체:
```tsx
import { CharacterCard } from '@/components/CharacterCard'
import { PillarsTable } from '@/components/PillarsTable'
import type { Character } from '@/lib/character'
import type { MbtiPrediction } from '@/lib/mbti'
import type { Person } from '@/lib/person'

type Props = {
  innerRef: React.RefObject<HTMLDivElement | null>
  person: Person
  character: Character
  chips: string[]
  prediction: MbtiPrediction
}

export function ShareCard({ innerRef, person, character, chips, prediction }: Props) {
  const { pillars, solar } = person
  return (
    // 공유 이미지는 4:5 세로 비율을 최소로 하고 내용이 많으면 늘어난다
    <div ref={innerRef} className="flex flex-col gap-4 rounded-2xl border bg-background p-5" style={{ aspectRatio: '4 / 5' }}>
      <div className="text-xs text-muted-foreground">{solar.y}.{solar.m}.{solar.d} (양력 기준)</div>
      <CharacterCard character={character} chips={chips} />
      <div className="space-y-2">
        <div className="flex items-center justify-between text-sm font-medium">
          <span>사주가 말하는 MBTI 경향</span>
          <span>{prediction.type}</span>
        </div>
        {prediction.axes.map((x) => {
          const pctSecond = 100 - x.pctFirst
          return (
            <div key={x.axis} className="flex items-center gap-2 text-xs">
              <span className="w-12 tabular-nums">{x.letters[0]} {x.pctFirst}%</span>
              <div className="flex h-1.5 flex-1 overflow-hidden rounded-full bg-muted">
                <div className="bg-sky-500" style={{ width: `${x.pctFirst}%` }} />
                <div className="bg-amber-500" style={{ width: `${pctSecond}%` }} />
              </div>
              <span className="w-12 text-right tabular-nums">{pctSecond}% {x.letters[1]}</span>
            </div>
          )
        })}
      </div>
      <PillarsTable pillars={pillars} />
    </div>
  )
}
```

- [ ] **Step 4: `Result` 재구성**

`saju_web/src/components/Result.tsx` 전체 교체:
```tsx
import { useMemo, useRef, useState } from 'react'
import { Button } from '@/components/ui/button'
import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs'
import { ShareCard } from '@/components/ShareCard'
import { MbtiCard } from '@/components/MbtiCard'
import { ReadingCard } from '@/components/ReadingCard'
import { SituationBubbles } from '@/components/SituationBubbles'
import { YearFortuneCard } from '@/components/YearFortuneCard'
import { PartnerRecommend } from '@/components/PartnerRecommend'
import { CompatPanel } from '@/components/CompatPanel'
import { buildCharacter } from '@/lib/character'
import { interpret, yearFortune } from '@/lib/interpret'
import { predictMbti } from '@/lib/mbti'
import { savePng, sharePng } from '@/lib/exportCard'
import type { Person } from '@/lib/person'

export function Result({ me, onReset }: { me: Person; onReset: () => void }) {
  const cardRef = useRef<HTMLDivElement>(null)
  const [busy, setBusy] = useState(false)
  const [error, setError] = useState(false)
  const year = useMemo(() => yearFortune(me.analysis, new Date().getFullYear()), [me])
  const sections = useMemo(() => interpret(me.pillars, me.analysis), [me])
  const prediction = useMemo(() => predictMbti(me.analysis), [me])
  const character = useMemo(
    () => buildCharacter({
      dayGan: me.pillars.day.gan,
      dayZhi: me.pillars.day.zhi,
      group: me.analysis.dominantGroup,
      strength: me.analysis.strength,
      dayElement: me.analysis.dayElement,
    }),
    [me],
  )

  const run = async (fn: (n: HTMLElement) => Promise<void>) => {
    if (!cardRef.current) return
    setBusy(true)
    setError(false)
    try {
      await fn(cardRef.current)
    } catch (e) {
      if ((e as Error).name !== 'AbortError') setError(true)
    } finally {
      setBusy(false)
    }
  }

  return (
    <div className="space-y-4">
      <Tabs defaultValue="reading">
        <TabsList className="grid w-full grid-cols-2">
          <TabsTrigger value="reading">풀이</TabsTrigger>
          <TabsTrigger value="compat">궁합</TabsTrigger>
        </TabsList>

        <TabsContent value="reading" className="space-y-3">
          <ShareCard innerRef={cardRef} person={me} character={character} chips={sections[0].chips} prediction={prediction} />
          <div className="flex gap-2">
            <Button className="flex-1" disabled={busy} onClick={() => run(savePng)}>이미지 저장</Button>
            <Button className="flex-1" variant="outline" disabled={busy} onClick={() => run(sharePng)}>공유</Button>
          </div>
          {error && <p className="text-sm text-destructive">이미지를 만들지 못했습니다. 다시 시도해 주세요.</p>}
          <MbtiCard prediction={prediction} />
          {sections.map((s) => <ReadingCard key={s.title} section={s} />)}
          <SituationBubbles bubbles={character.bubbles} />
          <YearFortuneCard y={year} />
        </TabsContent>

        <TabsContent value="compat" className="space-y-3">
          <PartnerRecommend me={me} />
          <CompatPanel me={me} />
        </TabsContent>
      </Tabs>
      <Button variant="ghost" className="w-full" onClick={onReset}>다시 입력</Button>
    </div>
  )
}
```

`npm run dev`를 백그라운드로 띄우고(포트는 출력된 것 사용, 자기가 띄운 PID만 종료) 내장 브라우저로 양력 1990/3/15 10:30 여성을 입력해 DOM으로 확인한다.

- MBTI 카드: 제목 옆 배지 `ISFJ`(분포 검증으로 `MBTI_GAIN`이 바뀌어도 유형 글자는 같다), `사주 기운을 MBTI 축으로 번역한 경향입니다.` 문구, 4축 막대, 각 축 `근거 보기`를 펼치면 근거와 조언 문장이 보인다.
- 비교: E, S, T, P를 차례로 누르면 벤다이어그램이 나타나고 `1/4 일치`(S만 일치)와 설명 3줄이 보인다. `다시 고르기`로 초기화된다.
- 풀이 카드 5개: 각각 요약, 칩 3개, 불릿 3줄, `더 보기`(기존 줄글). 오행 균형·사주 구조 카드에 막대가 있다.
- 말풍선 3개(연애할 때 / 일할 때 / 스트레스 받을 때), 올해 세운 카드.
- `시간을 몰라요`로 입력해도 모든 카드가 오류 없이 나온다. 콘솔 오류 없음.
- 이미지 저장: 내장 브라우저에서는 `toPng`가 `requestAnimationFrame`에 걸려 멈춘다. 이전처럼 테스트용 shim(`requestAnimationFrame = cb => setTimeout(cb, 0)`)을 페이지에서만 주입해 PNG data URL 생성만 확인하고 shim은 커밋하지 않는다. 실제 모양은 확인하지 못했다고 보고한다.
